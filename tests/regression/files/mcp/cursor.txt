import { McpServer } from "@modelcontextprotocol/server";
import { serveStdio } from "@modelcontextprotocol/server/stdio";
import * as z from "zod/v4";
import path from "node:path";
import os from "node:os";
import { readdir } from "node:fs/promises";

const DOSSIER_AUTORISE = path.join(os.homedir(), "projets");

function cheminAutorise(...parties: string[]): string | undefined {
  const chemin = path.resolve(DOSSIER_AUTORISE, ...parties);
  const relatif = path.relative(DOSSIER_AUTORISE, chemin);
  return relatif.startsWith("..") || path.isAbsolute(relatif) ? undefined : chemin;
}

function createServer(): McpServer {
  const server = new McpServer({ name: "projets-locaux", version: "1.0.0" });

  server.registerTool(
    "lister_projets",
    {
      title: "Lister les projets locaux",
      description: "Retourne la liste des dossiers présents dans ~/projets.",
      inputSchema: z.object({}),
    },
    async () => {
      try {
        const projets = await readdir(DOSSIER_AUTORISE, { withFileTypes: true });
        const dossiers = projets
          .filter((e) => e.isDirectory())
          .map((e) => e.name);
        return {
          content: [
            {
              type: "text",
              text: dossiers.length > 0 ? dossiers.join("\n") : "Aucun projet trouvé dans ~/projets.",
            },
          ],
        };
      } catch (error) {
        console.error("Erreur lors de la lecture du dossier ~/projets :", error);
        return {
          isError: true,
          content: [
            {
              type: "text",
              text: "Impossible de lire les projets dans ~/projets.",
            },
          ],
        };
      }
    },
  );

  server.registerTool(
    "lire_claude_md",
    {
      title: "Lire le fichier CLAUDE.md d’un projet",
      description:
        "Lit le contenu du fichier CLAUDE.md d’un projet spécifique situé dans ~/projets. Utile pour obtenir les instructions ou notes spécifiques à un projet.",
      inputSchema: z.object({
        nom_projet: z.string().describe("Nom du dossier du projet (ex: mon-projet)"),
      }),
    },
    async ({ nom_projet }) => {
      const cheminFichier = cheminAutorise(nom_projet, "CLAUDE.md");
      if (!cheminFichier) {
        return {
          isError: true,
          content: [
            {
              type: "text",
              text: `Accès refusé : le projet "${nom_projet}" n’est pas dans ~/projets ou le fichier CLAUDE.md est introuvable.`,
            },
          ],
        };
      }

      try {
        const { readFile } = await import("node:fs/promises");
        const contenu = await readFile(cheminFichier, "utf-8");
        return {
          content: [
            {
              type: "text",
              text: contenu,
            },
          ],
        };
      } catch (error) {
        console.error(`Erreur lors de la lecture de ${cheminFichier} :`, error);
        return {
          isError: true,
          content: [
            {
              type: "text",
              text: `Impossible de lire le fichier CLAUDE.md pour le projet "${nom_projet}".`,
            },
          ],
        };
      }
    },
  );

  return server;
}

void serveStdio(createServer);


=== projets-locaux/README.md (modele)
# Connecteur MCP « projets-locaux »

Ce connecteur permet d’interagir avec les projets locaux situés dans le dossier `~/projets`. Il offre deux outils en lecture seule :
- Lister les projets présents dans ce dossier.
- Lire le contenu du fichier `CLAUDE.md` d’un projet spécifique.

## Installation

1. Exécute `npm install` pour installer les dépendances.
2. Compile le code avec `npm run build`. Cela génère `dist/index.js`, le fichier exécuté par l’IA.
3. Teste les outils via `npm run inspect`.
4. Pour configurer ce connecteur dans Cursor, fusionne le contenu du fichier `config/cursor-mcp.json` dans le fichier `.cursor/mcp.json` du projet ou `~/.cursor/mcp.json`.
5. Remplace `/CHEMIN/ABSOLU/VERS` par le chemin réel vers le dossier du connecteur.

## Outils

### lister_projets
Retourne la liste des dossiers présents dans ~/projets.

### lire_claude_md
Lit le contenu du fichier `CLAUDE.md` d’un projet spécifique situé dans ~/projets.
