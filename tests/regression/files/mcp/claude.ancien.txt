import { McpServer } from "@modelcontextprotocol/server";
import { serveStdio } from "@modelcontextprotocol/server/stdio";
import * as z from "zod/v4";
import path from "node:path";
import os from "node:os";
import fs from "node:fs/promises";

const DOSSIER_AUTORISE = path.join(os.homedir(), "projets");

function cheminAutorise(...parties: string[]): string | undefined {
  const chemin = path.resolve(DOSSIER_AUTORISE, ...parties);
  const relatif = path.relative(DOSSIER_AUTORISE, chemin);
  return relatif.startsWith("..") || path.isAbsolute(relatif) ? undefined : chemin;
}

function createServer(): McpServer {
  const server = new McpServer({ name: "projets-locaux", version: "1.0.0" });

  server.registerTool(
    "lister_projets",
    {
      title: "Lister les projets",
      description: "Liste tous les répertoires dans le dossier ~/projets. Utile pour connaître les projets disponibles.",
      inputSchema: z.object({}),
    },
    async () => {
      try {
        const chemin = cheminAutorise();
        if (!chemin) {
          return { isError: true, message: "Accès refusé au dossier autorisé." };
        }
        const fichiers = await fs.readdir(chemin);
        const projets = fichiers.filter(fichier => 
          fs.stat(path.join(chemin, fichier)).then(stat => stat.isDirectory())
        );
        const resultats = await Promise.all(projets.map(async (nom) => {
          const cheminProjet = cheminAutorise(nom);
          if (!cheminProjet) return null;
          try {
            const stats = await fs.stat(cheminProjet);
            return {
              nom,
              dateCreation: stats.birthtime.toISOString(),
            };
          } catch (e) {
            console.error(`Erreur lors de la lecture des stats du projet ${nom}:`, e);
            return { nom, dateCreation: null };
          }
        }));
        return { content: [{ type: "text", text: JSON.stringify(resultats.filter(Boolean)) }] };
      } catch (error) {
        console.error("Erreur lors de la liste des projets:", error);
        return { isError: true, message: "Impossible de lire les projets." };
      }
    },
  );

  server.registerTool(
    "lire_claude_md",
    {
      title: "Lire le CLAUDE.md d'un projet",
      description: "Lit le contenu du fichier CLAUDE.md d'un projet spécifique. Utile pour obtenir les instructions ou notes du projet.",
      inputSchema: z.object({
        nom_projet: z.string().describe("Nom du dossier du projet (ex: 'mon-projet')"),
      }),
    },
    async ({ nom_projet }) => {
      try {
        const cheminProjet = cheminAutorise(nom_projet);
        if (!cheminProjet) {
          return { isError: true, message: "Accès refusé au projet spécifié." };
        }
        const cheminFichier = path.join(cheminProjet, "CLAUDE.md");
        const contenu = await fs.readFile(cheminFichier, "utf-8");
        return { content: [{ type: "text", text: contenu }] };
      } catch (error) {
        console.error(`Erreur lors de la lecture du fichier CLAUDE.md pour le projet ${nom_projet}:`, error);
        return { isError: true, message: `Impossible de lire le fichier CLAUDE.md pour le projet ${nom_projet}.` };
      }
    },
  );

  return server;
}

void serveStdio(createServer);


=== projets-locaux/README.md (modele)
# Connecteur MCP « projets-locaux »

Ce connecteur permet d'interagir avec les projets stockés dans le dossier `~/projets`.

## Outils

### lister_projets
Liste tous les répertoires dans le dossier `~/projets`. Utile pour connaître les projets disponibles.

### lire_claude_md
Lit le contenu du fichier `CLAUDE.md` d'un projet spécifique. Utile pour obtenir les instructions ou notes du projet.

## Installation

1. npm install
2. npm run build (crée dist/index.js, le fichier lancé par l'IA)
3. npm run inspect (tester les outils dans l'inspecteur MCP)
4. Configuration : fusionner le contenu du dossier `config/` dans la configuration de Claude :
   - Pour Claude Code : fusionner dans `.mcp.json` à la racine du projet.
   - Pour Claude Desktop : fusionner dans `claude_desktop_config.json`.
5. Remplacer `/CHEMIN/ABSOLU/VERS` par le chemin réel du dossier du connecteur.

À vérifier :
- Le dossier `~/projets` existe bien et est accessible.
- Les fichiers `CLAUDE.md` sont présents dans les projets.
- La configuration MCP est correctement fusionnée dans le fichier de l'IA cible.
