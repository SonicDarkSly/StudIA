=== .claude/hooks/safe_bash.sh (modele)
#!/usr/bin/env bash
set -euo pipefail

# Lecture de l'entrée standard (JSON)
input=$(cat)

# Extraction de la commande Bash
command=$(echo "$input" | jq -r '.tool_input.command // empty')

# Vérification si la commande contient des motifs dangereux
if [[ "$command" == *"rm -rf"* ]] || [[ "$command" == *"git push --force"* ]]; then
  echo "⚠️  Exécution bloquée : cette commande est potentiellement destructive." >&2
  exit 2
fi

# Si aucune condition n'est remplie, on laisse la commande s'exécuter
exit 0
